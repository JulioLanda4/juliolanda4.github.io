\newcommand{\cvheader}[2]{%
  \vspace*{-0.35cm}%
  \noindent
  \begin{minipage}[t]{0.64\textwidth}
    {\color{cvnavy}\fontsize{26}{28}\selectfont\bfseries\MakeUppercase{#1}}\par
    \vspace{0.08cm}
    {\color{cvteal}\fontsize{10.5}{12}\selectfont #2}
  \end{minipage}
  \hfill
  \begin{minipage}[t]{0.27\textwidth}
    \raggedleft
    {\color{cvnavy}\fontsize{28}{28}\selectfont\bfseries JL}
  \end{minipage}
  \par\vspace{0.42cm}
}

\newcommand{\cvpaintsidebar}{%
  \begin{tikzpicture}[remember picture,overlay]
    \fill[cvnavy]
      ([xshift=-0.36\paperwidth]current page.north east)
      rectangle
      (current page.south east);
    \fill[cvteal]
      ([xshift=-0.36\paperwidth]current page.north east)
      rectangle
      ([xshift=-0.352\paperwidth]current page.south east);
    \node[
      circle,
      fill=cvsoft,
      draw=cvteal,
      line width=1pt,
      minimum size=1.62cm,
      text=cvnavy,
      font=\bfseries\fontsize{18}{18}\selectfont
    ] at ([xshift=-0.18\paperwidth,yshift=-1.48cm]current page.north east) {JL};
  \end{tikzpicture}%
}
